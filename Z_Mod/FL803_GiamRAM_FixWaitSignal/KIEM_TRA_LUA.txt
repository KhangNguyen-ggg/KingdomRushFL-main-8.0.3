local files = {"all/systems.lua", "all/game.lua", "main.lua"}
local all_ok = true
for _, f in ipairs(files) do
  local chunk, err = loadfile(f)
  if chunk then
    print("OK  : " .. f)
  else
    all_ok = false
    print("LOI : " .. f .. " -> " .. tostring(err))
  end
end
print(all_ok and "TAT CA OK - co the choi game" or "CO LOI CU PHAP - khong choi, rollback bang BACKUP")
